cd - можно перемещаться по файлам системы
git init - инициализирует пустой гит репозиторий в месте где ты находишься
echo > - создает файл
dir - показывает текущие файлы в папке где ты находишься
nano - позволяет изменять данные файла и сохранять их, так же изменять названия файла
cat - показывает содержимое файла
git add - добавляет файл в репозиторий
git status - проверяет статус файлов в текущей папке (как я понял)
git commit -m "new index.html" коммитит файлы в репозиторий
